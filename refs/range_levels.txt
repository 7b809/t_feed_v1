OR candle = first candle of the day (sorted ascending by time)

Open  = first.open
High  = first.high
Low   = first.low
Close = first.close

R1 = High
S1 = Low
AVG = (R1 + S1) / 2
    = (High + Low) / 2

R1_Diff = R1 - AVG
        = High - ((High + Low) / 2)
        = (High - Low) / 2

S1_Diff = AVG - S1
        = ((High + Low) / 2) - Low
        = (High - Low) / 2

R2 = R1 + R1_Diff
   = High + ((High - Low) / 2)

R3 = R2 + R1_Diff
   = High + (High - Low)

S2 = S1 - S1_Diff
   = Low - ((High - Low) / 2)

S3 = S2 - S1_Diff
   = Low - (High - Low)


   
================================================================
SAMPLE EXAMPLE
================================================================

INPUT (first candle of the day):
    Open  = 21960
    High  = 22000
    Low   = 21900
    Close = 21980

STEP 1 — Range:
    Range = High - Low
    Range = 22000 - 21900
    Range = 100

STEP 2 — Average:
    Average = (High + Low) / 2
    Average = (22000 + 21900) / 2
    Average = 43900 / 2
    Average = 21950

STEP 3 — Deltas:
    R1_Diff = High - Average
    R1_Diff = 22000 - 21950
    R1_Diff = 50

    S1_Diff = Average - Low
    S1_Diff = 21950 - 21900
    S1_Diff = 50

    (Both equal Range / 2 = 50  ✓)

STEP 4 — Levels:

    R1 = High
    R1 = 22000

    R2 = High + R1_Diff
    R2 = 22000 + 50
    R2 = 22050

    R3 = High + (R1_Diff * 2)
    R3 = 22000 + 100
    R3 = 22100

    AVG = (High + Low) / 2
    AVG = 21950

    S1 = Low
    S1 = 21900

    S2 = Low - S1_Diff
    S2 = 21900 - 50
    S2 = 21850

    S3 = Low - (S1_Diff * 2)
    S3 = 21900 - 100
    S3 = 21800

FINAL OUTPUT:
    +-------+---------+
    | Level | Value   |
    +-------+---------+
    | R3    | 22100   |
    | R2    | 22050   |
    | R1    | 22000   |
    | AVG   | 21950   |
    | S1    | 21900   |
    | S2    | 21850   |
    | S3    | 21800   |
    +-------+---------+
